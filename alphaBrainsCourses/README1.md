sed 's/unix/linux/' geekfile.txt

sed 's/unix/linux/2' geekfile.txt       # replaces the second/nth occurance only

sed 's/unix/linux/g' geekfile.txt       # replaces all occurance global

sed 's/unix/linux/3g' geekfile.txt      # Combining both global and nth occurance

echo "Welcome To The Geek Stuff" | sed 's/\(\b[A-Z]\)/\(\1\)/g'     # Parenthesize First Character of Each Word

sed '3 s/unix/linux/' geekfile.txt      # Replacing String on a Specific Line Number

sed 's/unix/linux/p' geekfile.txt       # Duplicating the Replaced Line with /p flag

sed -n 's/unix/linux/p' geekfile.txt    # Printing Only the Replaced Lines the -n option suppresses the duplicate rows generated by the /p flag and prints the replaced lines only one time.

# Replacing String on a Range of Lines
sed '1,3 s/unix/linux/' geekfile.txt
sed '2,$ s/unix/linux/' geekfile.txt

# To Delete a particular line say n in this example
sed 'nd' filename.txt
sed '5d' filename.txt
# To Delete a last line
sed '$d' filename.txt
# To Delete line from range x to y
sed 'x,yd' filename.txt
sed '3,6d' filename.txt
# To Delete from nth to last line
sed 'nth,$d' filename.txt
sed '12,$d' filename.txt
# To Delete pattern matching line
sed '/pattern/d' filename.txt
sed '/abc/d' filename.txt
# Regular Expressions: Here, we have matched any word start with 'u' and replacing it with "Linux".
sed -r 's/\bu\w+/Linux/g' geekfile.txt
# Insert Text
sed '3i\new text' filename  # Insert text before line 3
sed '3a\new text' filename  # Insert text after line 3
sed '2i\This is a new line' geekfile.txt

# Option/Flag   What it does
-i              Edits the file in-place (overwrites the original file).
-n              Suppresses automatic printing; helpful when you only want to print specific lines.
g               Global flag; replaces all matches on a line instead of just the first one.
d               Delete flag; deletes the specified line or pattern match.
p               Print flag; typically paired with -n to print specific matches (e.g., sed -n '/pattern/p').
s               Substitution
c               Replace entire line
a               Append
